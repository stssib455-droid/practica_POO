cmake_minimum_required(VERSION 3.16)
project(Practical_01_Classes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
add_library(drone_core drone.cpp drone.h)
target_include_directories(drone_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(console_demo console_main.cpp)
target_link_libraries(console_demo PRIVATE drone_core)
add_executable(drone_tests drone_tests.cpp)
target_link_libraries(drone_tests PRIVATE drone_core)
include(CTest)
add_test(NAME drone_behavior COMMAND drone_tests)

add_library(flight_ui mainwindow.cpp mainwindow.h mainwindow.ui flight_scene.h flight_scene.cpp)
target_link_libraries(flight_ui PUBLIC drone_core Qt6::Widgets)
target_include_directories(flight_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(gui_tests gui_tests.cpp)
target_link_libraries(gui_tests PRIVATE flight_ui Qt6::Test)
add_test(NAME flight_interface COMMAND gui_tests)
qt_add_executable(Practical_01_Classes WIN32 main.cpp)
target_link_libraries(Practical_01_Classes PRIVATE flight_ui)
