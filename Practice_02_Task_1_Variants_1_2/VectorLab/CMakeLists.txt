cmake_minimum_required(VERSION 3.16)
project(VectorLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
add_library(vector_core vector3d.cpp vector3d.h)
target_include_directories(vector_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(vector_tests vector_tests.cpp)
target_link_libraries(vector_tests PRIVATE vector_core)
include(CTest)
add_test(NAME vector_operations COMMAND vector_tests)

add_library(vector_ui mainwindow.cpp mainwindow.h mainwindow.ui vector_scene.cpp vector_scene.h)
target_link_libraries(vector_ui PUBLIC vector_core Qt6::Widgets)
target_include_directories(vector_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(VectorLab WIN32 main.cpp)
target_link_libraries(VectorLab PRIVATE vector_ui)
add_executable(console_demo console_demo.cpp)
target_link_libraries(console_demo PRIVATE vector_core)
add_executable(gui_tests gui_tests.cpp)
target_link_libraries(gui_tests PRIVATE vector_ui Qt6::Test)
add_test(NAME vector_interface COMMAND gui_tests -o gui-tests.txt,txt)
